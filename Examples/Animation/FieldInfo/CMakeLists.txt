find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(testAnimationFieldInfo Validation.cpp)
target_link_libraries(testAnimationFieldInfo PRIVATE Qt_module)
if (NOT IGAME_BUILD_SHARED)
    target_compile_definitions(testAnimationFieldInfo PRIVATE IG_QT_COMPILE_EXPORT)
endif ()
if (MSVC)
    target_compile_options(testAnimationFieldInfo PRIVATE /utf-8)
endif ()
set_target_properties(testAnimationFieldInfo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
get_target_property(_info_qmake Qt5::qmake IMPORTED_LOCATION)
get_filename_component(_info_qt_bin "${_info_qmake}" DIRECTORY)
add_test(NAME Animation.FieldInfo.Descriptions COMMAND testAnimationFieldInfo descriptions)
add_test(NAME Animation.FieldInfo.ParameterDialog COMMAND testAnimationFieldInfo dialog)
set_tests_properties(Animation.FieldInfo.Descriptions Animation.FieldInfo.ParameterDialog PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORM_PLUGIN_PATH=${_info_qt_bin}/../plugins/platforms"
    WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" TIMEOUT 60)
